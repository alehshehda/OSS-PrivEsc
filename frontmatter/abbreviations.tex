\UnnumberedChapter{\abbrevname}

\begin{abbreviations}
    \item[ACL]  Access Control List (lista kontroli dostępu)
    \item[CVE]  Common Vulnerabilities and Exposures (baza znanych podatności bezpieczeństwa)
    \item[DAC]  Discretionary Access Control (uznaniowa kontrola dostępu)
    \item[GID]  Group Identifier (identyfikator grupy w systemach uniksowych)
    \item[MAC]  Mandatory Access Control (obowiązkowa kontrola dostępu)
    \item[PAM]  Pluggable Authentication Modules (modułowy mechanizm uwierzytelniania)
    \item[PoC]  Proof of Concept (dowód słuszności koncepcji / kod demonstracyjny eksploitu)
    \item[SGID] Set Group ID (bit uprawnień wykonania z prawami grupy właściciela pliku)
    \item[SUID] Set User ID (bit uprawnień wykonania z prawami właściciela pliku)
    \item[UID]  User Identifier (identyfikator użytkownika w systemach uniksowych)
\end{abbreviations}
